```zig
// v1.0 使用示例
const poly = @import("poly-sdk-zig");

pub fn main() !void {
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();
    const allocator = gpa.allocator();

    // 1. 创建客户端
    var client = poly.ClobClient.init(allocator, .{});
    defer client.deinit();

    // 2. 公共 API (L0)
    const markets = try client.getMarkets(.{});
    defer markets.deinit();

    // 3. L1 认证 - 获取 API 凭证
    const wallet = try poly.Wallet.fromPrivateKeyHex("0x...");
    client.setWallet(&wallet);
    var creds = try client.createOrDeriveApiKey();
    defer creds.deinit();
    client.setApiCreds(&creds);

    // 4. L2 认证 - 创建订单
    const order = try client.createOrder(.{
        .token_id = markets.value[0].tokens[0].token_id,
        .price = try poly.Decimal.fromString("0.65"),
        .size = try poly.Decimal.fromString("100"),
        .side = .buy,
    });
    _ = try client.postOrder(&order, .GTC);
}
```
